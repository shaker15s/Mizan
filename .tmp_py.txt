import sys
sys.path.insert(0, '/usr/lib/python3/dist-packages')
from odoo import api, SUPERUSER_ID
from odoo.modules.registry import Registry
from odoo.tools.config import config
config.parse_config(['-c','/etc/odoo/odoo.conf'])
registry = Registry('poc_test')
with registry.cursor() as cr:
    env = api.Environment(cr, SUPERUSER_ID, {})
    user = env['res.users'].search([('login','=','readonly_user@test')], limit=1)
    groups = env['res.groups'].search([('name','in',['Allowed','Role / User','Show Accounting Features - Readonly'])])
    user.group_ids = [(6,0,groups.ids)]
    env['res.groups'].search([])  # touch groups to trigger recompute
    print('readonly groups:', user.group_ids.mapped(lambda g: g.name))
    print('readonly all groups:', user.all_group_ids.ids)
